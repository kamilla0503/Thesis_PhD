\begin{frame}{Центральный вопрос}
  \large
  Что происходит с магнитным фазовым переходом, если спины живут не на фиксированной регулярной решетке, а на полимерной цепочке, геометрия которой сама является термодинамической степенью свободы?

  \vspace{1em}
  \normalsize
  \begin{columns}[T,onlytextwidth]
    \column{0.32\textwidth}
    \begin{block}{Спины}
      Изинг: \(s_i=\pm1\).\\
      XY: \(\theta_i\in(-\pi,\pi]\).
    \end{block}
    \column{0.32\textwidth}
    \begin{block}{Геометрия}
      Блуждание без самопересечений \(u\in\Usaw_N\).\\
      Цепочка может быть клубком или глобулой.
    \end{block}
    \column{0.32\textwidth}
    \begin{block}{Связь}
      Контакты меняют энергию спинов.\\
      Спины меняют статистику контактов.
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Маршрут на 60 минут}
  \small
  \begin{tabularx}{\textwidth}{@{}lX@{}}
    \toprule
    0--10 мин & Классические модели: Изинг, XY, БКТ, \(\Theta\)-переход полимера. \\
    10--18 мин & Разработанность темы: от магнитных полимеров к Potts, BEG и новым двумерным/двухблочным направлениям. \\
    18--30 мин & Модель и численные методы: MCMC, смена соединений, кластерный шаг, обмен репликами, перевзвешивание. \\
    30--42 мин & Изинг на БСП: совместный непрерывный структурно-магнитный переход в 2D. \\
    42--54 мин & XY на БСП: разделение переходов в 2D и признаки первого рода в 3D. \\
    54--60 мин & Синтез, новизна, положения, будущие планы. \\
    \bottomrule
  \end{tabularx}

  \vspace{0.8em}
  \begin{block}{Научная история}
    Не ``практическая значимость'', а механизм: как симметрия спина, размерность и динамическая геометрия вместе определяют фазовую картину.
  \end{block}
\end{frame}

\section{Классические модели и переходы}

\begin{frame}{Модель Изинга: дискретная симметрия}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
    \small
    \[
      H_{\mathrm{Ising}}
      = -J\sum_{\langle i,j\rangle} s_i s_j,\qquad
      s_i=\pm1 .
    \]
    \begin{itemize}
      \item Параметр порядка: \(m=N^{-1}\sum_i s_i\).
      \item В 2D на квадратной решетке есть точное решение.
      \item При \(T<T_c\): ферромагнитный порядок.
      \item При \(T>T_c\): парамагнитный режим.
    \end{itemize}
    \column{0.48\textwidth}
    \centering
    \begin{tikzpicture}[scale=0.75]
      \foreach \x in {0,...,5}{
        \foreach \y in {0,...,4}{
          \draw[gray!35] (\x,\y) -- ++(1,0);
          \draw[gray!35] (\x,\y) -- ++(0,1);
        }
      }
      \foreach \x in {0,...,6}{
        \foreach \y in {0,...,5}{
          \pgfmathtruncatemacro{\up}{mod(\x+\y,3)}
          \ifnum\up=0
            \draw[very thick,blue!70!black,-{Latex[length=2mm]}] (\x,\y-0.22) -- (\x,\y+0.22);
          \else
            \draw[very thick,red!70!black,-{Latex[length=2mm]}] (\x,\y+0.22) -- (\x,\y-0.22);
          \fi
        }
      }
      \node[font=\scriptsize] at (3,-0.7) {локальные взаимодействия ближайших соседей};
    \end{tikzpicture}
  \end{columns}
  \source{E. Ising, Z. Phys. 31, 253--258 (1925); L. Onsager, Phys. Rev. 65, 117--149 (1944).}
\end{frame}

\begin{frame}{Классический переход Изинга}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{Что видно в численном эксперименте}
      \begin{itemize}
        \item Пересечения кумулянта Биндера \(U_4\).
        \item Рост пика теплоемкости \(c_V\).
        \item Масштабирование намагниченности и восприимчивости.

\end{frame}

\begin{frame}{XY-модель: непрерывная симметрия}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
    \small
    \[
      H_{XY}=-J\sum_{\langle i,j\rangle}\cos(\theta_i-\theta_j),
      \qquad
      \vec s_i=(\cos\theta_i,\sin\theta_i).
    \]
    \begin{itemize}
      \item Симметрия \(O(2)\), параметр порядка в конечной системе:
      \[
        m=\frac{1}{N}\sqrt{\left(\sum_i\cos\theta_i\right)^2+
        \left(\sum_i\sin\theta_i\right)^2}.
      \]
      \item В 2D обычного дальнего порядка нет.
      \item Вместо обычного перехода возникает БКТ-сценарий.
    \end{itemize}
    \column{0.48\textwidth}
    \centering
    \begin{tikzpicture}[scale=0.75]
      \foreach \x in {0,...,6}{
        \foreach \y in {0,...,5}{
          \fill[gray!25] (\x,\y) circle (0.025);
          \pgfmathsetmacro{\ang}{25*\x+35*\y+10*sin(30*\x)}
          \draw[very thick,teal!70!black,-{Latex[length=2mm]}]
            (\x,\y) -- ++(\ang:0.35);
        }
      }
      \node[font=\scriptsize] at (3,-0.7) {угловые степени свободы};
    \end{tikzpicture}
  \end{columns}
  \source{J. M. Kosterlitz and D. J. Thouless, J. Phys. C 6, 1181 (1973); В. Л. Березинский, ЖЭТФ (1970, 1971).}
\end{frame}

\begin{frame}{БКТ-переход: не обычная критическая точка}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
    \begin{block}{Физическая картина}
      \begin{itemize}
        \item При низких \(T\): связанные пары вихрь-антивихрь.
        \item При высоких \(T\): свободные вихри разрушают квазидальний порядок.
        \item Корреляционная длина расходится экспоненциально, а не степенно.
      \end{itemize}
    \end{block}
    \[
      G(r)\sim r^{-\eta},\qquad
      \eta(T_{\mathrm{BKT}})=\frac14.
    \]
    \[
      \xi \sim \exp\!\left(\frac{b}{\sqrt{|T-T_{\mathrm{BKT}}|}}\right).
    \]

\begin{frame}{Полимер: блуждание без самопересечений}
  \begin{columns}[T,onlytextwidth]
    \column{0.5\textwidth}
    \small
    Геометрия линейного полимера длины \(N\) задается БСП:
    \[
      u=(r_1,\dots,r_N),\qquad r_i\ne r_j \ (i\ne j).
    \]
    Размер цепи:
    \[
      \langle R_N^2\rangle \sim N^{2\nu}.
    \]
    \begin{tabular}{@{}lcc@{}}
      \toprule
      режим & 2D & 3D \\
      \midrule
      хорош. растворитель & \(3/4\) & \(\simeq0.588\) \\
      \(\Theta\)-точка & \(4/7\) & \(1/2\) \\
      глобула & \(1/2\) & \(1/3\) \\
      \bottomrule
    \end{tabular}
    \column{0.5\textwidth}
    \centering
    \includegraphics[width=0.6\linewidth]{homopolymer.png}
    \vspace{0.5em}

    \scriptsize
    Переход клубок-глобула:
    \[
      \frac{\langle R_N^2\rangle}{N^{2\nu_\Theta}}
      =
      \R\!\left((J-J_\Theta)N^{\phi_\Theta}\right).
    \]
  \end{columns}
  \source{P.-G. de Gennes, Scaling Concepts in Polymer Physics (1979); Duplantier and Saleur, Phys. Rev. Lett. 59, 539 (1987).}
\end{frame}

\begin{frame}{Почему объединение моделей не является суммой частей}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{На регулярной решетке}
      Связи фиксированы: каждое изменение спиновой конфигурации происходит на одном и том же графе.
    \end{block}
    \begin{block}{На БСП}
      Связи флуктуируют: число топологических контактов само является случайной величиной и зависит от \(J\).
    \end{block}
    \column{0.52\textwidth}
    \centering

      \draw[-{Latex[length=2mm]},thick] (spin) -- node[right,font=\tiny] {стабилизирует} (contacts);

    \end{tikzpicture}
  \end{columns}
  \vspace{0.5em}
  \begin{block}{Рабочая гипотеза}
    Симметрия спина определяет, насколько сильно магнитное упорядочение привязано к компактизации цепи.
  \end{block}
\end{frame}

\section{Разработанность темы}

\begin{frame}{Магнитные полимеры: две постановки беспорядка}
  \small
  \[
    Z_N=\sum_{u\in\Usaw_N}\sum_{s\in\mathcal S}
    \exp[-H(s,u)].
  \]
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{Замороженная геометрия}
      \begin{itemize}
        \item Сначала генерируется ансамбль БСП.
        \item Потом на каждой фиксированной конфигурации моделируются спины.
        \item Геометрия не реагирует на магнитное упорядочение.
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Разбавленный беспорядок}
      \begin{itemize}
        \item Геометрия и спины входят в один ансамбль.
        \item Контакты возникают и исчезают вместе со спиновыми флуктуациями.
        \item Это постановка данной диссертации.
      \end{itemize}
    \end{block}
  \end{columns}
  \source{Chakrabarti and Bhattacharya, J. Phys. C 16, L1025 (1983); Chakrabarti, Maggs and Stinchcombe, J. Phys. A 18, L373 (1985).}
\end{frame}

\section{Модели и методы}

\begin{frame}{Общая постановка}
  \small
  Полимерная конфигурация \(u\) задает множество взаимодействующих пар \(\E(u)\): это пары мономеров, которые являются соседями на регулярной решетке.
  \[
    Z_N(J)=\sum_{u\in\Usaw_N}\sum_{s\in\mathcal S}
    \exp\!\left[J\sum_{(i,j)\in\E(u)} e_{ij}\right].
  \]
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{Модель Изинга}
      \[
        e_{ij}=s_i s_j,\qquad s_i=\pm1.
      \]
      Дискретная симметрия, обычный параметр порядка \(m\).
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{XY-модель}
      \[
        e_{ij}=\cos(\theta_i-\theta_j),\qquad \theta_i\in(-\pi,\pi].
      \]
      Непрерывная симметрия, в 2D ожидается БКТ-подобная логика.
    \end{block}

\begin{frame}{Расширенная выборка: обмен репликами}
  \small
  Одного фиксированного \(J\) недостаточно, когда есть узкая критическая область или признаки первого рода. Поэтому использовались параллельное темперирование и мультигистограммное перевзвешивание.
  \[
    A_{\mathrm{swap}}=
    \min\left\{1,
    \exp\left[(J_m-J_{m+1})(E(u_m)-E(u_{m+1}))\right]\right\}.
  \]
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{Что дает обмен репликами}
      \begin{itemize}
        \item лучшее прохождение энергетических барьеров;
        \item один запуск покрывает окно значений \(J\);
        \item особенно полезен при бимодальности.
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Что дает перевзвешивание}
      \begin{itemize}
        \item гладкие зависимости \(U_4(J)\), \(R^2(J)\), \(c_V(J)\);
        \item уточнение пересечений;
        \item проверка опубликованных выводов на новых данных.
      \end{itemize}
    \end{block}
  \end{columns}
  \source{Swendsen and Wang, PRL 57, 2607 (1986); Ferrenberg and Swendsen, PRL 63, 1195 (1989).}
\end{frame}

    Распределения & Одномодальные & Бимодальные \\
    Вывод & Непрерывный, БКТ-подобный магнитный переход & Признаки перехода первого рода \\

  \vspace{0.8em}
  \begin{block}{Главный урок}
    Непрерывная симметрия сама по себе не задает один универсальный сценарий: размерность и динамическая контактная сеть решают, будет ли геометрия и магнетизм разделены или сцеплены.
  \end{block}
\end{frame}

\section{Синтез и вклад}

\begin{frame}{Сводная фазовая история}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.5\textwidth}
    \begin{block}{Изинг 2D}
      Дискретное упорядочение и компактизация происходят в одной критической области:
      \[
        J_\Theta\simeq J_{\mathrm{cr}}\simeq0.834.
      \]
    \end{block}
    \begin{block}{XY 2D}
      Геометрия перестраивается раньше, чем возникает БКТ-подобная магнитная область:
      \[
        J_\Theta\simeq1.36,\quad J_{\mathrm{BKT}}^{\mathrm{eff}}\simeq1.42.
      \]
    \end{block}
    \column{0.5\textwidth}
    \begin{block}{XY 3D}
      Контакты и намагниченность сильно сцеплены; распределения показывают двухфазность:
      \[
        J\simeq0.87\text{--}0.91.
      \]
    \end{block}
    \begin{block}{Общий вывод}
      Динамическая геометрия не просто ``шум'' для спиновой модели. Она активный участник фазового перехода.
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Научная новизна}
  \small
  \begin{itemize}
    \item Разработан и реализован численный подход для совместного моделирования спиновых и геометрических степеней свободы на БСП.
    \item Для модели Изинга на 2D БСП подтверждена совместность структурного и магнитного переходов и уточнены оценки с использованием обмена репликами и перевзвешивания.
    \item Впервые численно исследована XY-модель на БСП в 2D и 3D.
    \item Показано, что XY-модель сохраняет геометрическую \(\Theta\)-универсальность родительского полимера, но меняет связь между геометрией и магнитным порядком.
    \item Для 2D XY получена картина раздельных переходов и БКТ-подобной магнитной области.
    \item Для 3D XY обнаружены устойчивые признаки перехода первого рода: отрицательные провалы кумулянтов и бимодальные распределения.
  \end{itemize}
\end{frame}

\begin{frame}{Положения, выносимые на защиту: Изинг}
  \small
  \begin{enumerate}
    \item Численно исследована модель Изинга на динамических структурах, генерируемых блужданиями без самопересечений на регулярной решетке.
    \item Для квадратной решетки структурный переход клубок-глобула и магнитный переход происходят в одной критической области.
    \item Магнитный переход в двумерной модели Изинга на БСП имеет непрерывный характер.
    \item Геометрический критический показатель в области перехода наследуется от родительской модели гомополимера: \(\nu_\Theta=4/7\).
  \end{enumerate}
\end{frame}

\begin{frame}{Положения, выносимые на защиту: XY}
  \small
  \begin{enumerate}
    \item Впервые численно исследована XY-модель на блужданиях без самопересечений на квадратной и кубической решетках.
    \item В обеих размерностях обнаружен структурный переход клубок-глобула с геометрическим показателем родительской полимерной модели: \(4/7\) в 2D и \(1/2\) в 3D.
    \item В двумерной XY-модели структурный и магнитный переходы разделены по параметру взаимодействия; магнитная область согласована с БКТ-подобной картиной.
    \item В трехмерной XY-модели магнитный переход и структурная перестройка демонстрируют признаки перехода первого рода.
  \end{enumerate}
\end{frame}

\begin{frame}{Что это говорит о механизме}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{Дискретные спины}
      Контакты быстро стабилизируют одинаковые \(s_i\). Поэтому в Изинге геометрия и магнетизм синхронизируются.
    \end{block}
    \begin{block}{Непрерывные спины в 2D}
      Угловые флуктуации позволяют цепи уже стать компактной, но не дают обычного дальнего порядка; появляется БКТ-подобная область.
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Непрерывные спины в 3D}
      Компактность резко увеличивает контактную сеть; магнитное упорядочение и геометрия начинают усиливать друг друга, что приводит к двухфазности.
    \end{block}
    \begin{block}{В одном предложении}
      Тип спина задает магнитную мягкость, а размерность задает цену компактизации.
    \end{block}
  \end{columns}
\end{frame}

\section{Идеи и будущие планы}

\begin{frame}{Будущее 1: дальнодействующая XY-модель}
  \scriptsize
  Ближайшее естественное расширение:
  \[
    H_{\mathrm{LR}}(u,\theta)=
    -J\sum_{i<j}K(r_{ij})\cos(\theta_i-\theta_j),
    \qquad
    K(r)\sim r^{-(d+\sigma)}.
  \]
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{Почему это важно}
      Короткодействующая модель использует только ближайшие пространственные контакты. Реальные магнитные или эпигенетические взаимодействия могут быть эффективнее описаны дальнодействующим ядром.
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Главный вопрос}
      При каком \(\sigma\) 2D XY на БСП перестает быть БКТ-подобной и переходит к обычному или первородному сценарию?
    \end{block}
  \end{columns}
  \centering
  \includegraphics[width=0.55\linewidth,height=0.34\textheight,keepaspectratio]{longrangeshortchains_3D_U4_No1stordersigns_delta_E.png}
\end{frame}

\begin{frame}{Будущее 2: clock-модели \(Z_q\)}
  \small
  \[
    \theta_i\in\left\{0,\frac{2\pi}{q},\dots,\frac{2\pi(q-1)}{q}\right\}.
  \]
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{Почему это хороший мост}
      \begin{itemize}
        \item \(q=2\): Изинг.
        \item \(q\to\infty\): XY.
        \item При промежуточных \(q\) можно проследить, где появляется разделение переходов.
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{План}
      Построить фазовую диаграмму в координатах \((q,J)\): линии \(\Theta\)-перехода, магнитного перехода и область совпадения/расщепления.
    \end{block}
  \end{columns}
  \begin{block}{Научный смысл}
    Это прямой тест гипотезы: непрерывность внутренней симметрии ослабляет сцепление геометрии и магнетизма.
  \end{block}
\end{frame}

\begin{frame}{Будущее 3: две цепочки, блоки и хроматин}
  \small
  Свежие работы двигаются к более сложной внутренней структуре: Potts/BEG-состояния, нейтральные метки, две фазы контактов, блоки с разными магнитными взаимодействиями.
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{Двухцепочечная постановка}
      Две БСП или две связанные части цепи могут иметь разные типы взаимодействий и конкурировать за контакты.
    \end{block}
    \begin{block}{Diblock-идея}
      Ferro-взаимодействие внутри блоков и antiferro-взаимодействие между блоками дают сегрегацию, смешанную глобулу и гибридные фазы.
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Что можно взять из диссертации}
      Уже есть инфраструктура: динамическая геометрия, контакты, обмен репликами, диагностика распределений и корреляций.
    \end{block}
    \begin{block}{Что нужно добавить}
      Новые ходы для двух цепей/блоков, контроль переплетения и контактные карты.
    \end{block}
  \end{columns}
  \source{Raiola, Locatelli, Marenduzzo and Orlandini, arXiv:2602.22382 (2026).}
\end{frame}

\begin{frame}{Будущее 4: вычислительная стратегия}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{Методические ограничения сейчас}
      \begin{itemize}
        \item длинные автокорреляции в глобуле;
        \item сильные конечномерные поправки;
        \item 3D требует намного больших \(N\);
        \item БКТ-диагностика на открытой динамической геометрии неполна.
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{План усиления}
      \begin{itemize}
        \item GPU/многорепличные расчеты;
        \item адаптивный выбор сетки \(J_m\);
        \item более точные распределения в 3D;
        \item новые наблюдаемые для топологических дефектов XY.
      \end{itemize}
    \end{block}
  \end{columns}
  \begin{block}{Практическая цель для следующих работ}
    Не просто увеличить статистику, а закрыть слабые места вывода: порядок 3D-перехода и статус БКТ-подобной области в 2D.
  \end{block}
\end{frame}

\begin{frame}{Заключение}
  \large
  Диссертация показывает, что спиновая модель на динамическом полимерном носителе не сводится ни к спиновой модели на решетке, ни к обычному гомополимеру.

  \vspace{0.8em}
  \normalsize
  \begin{itemize}
    \item В Изинге 2D геометрия и магнетизм идут вместе.
    \item В XY 2D они разделяются, а магнитная область выглядит БКТ-подобной.
    \item В XY 3D они снова сцепляются, но уже с признаками первого рода.
  \end{itemize}

  \vspace{0.8em}
  \begin{block}{Главный результат}
    Симметрия спиновых степеней свободы и размерность определяют, как динамическая геометрия превращает контактную сеть полимера в магнитный фазовый переход.
  \end{block}
\end{frame}
